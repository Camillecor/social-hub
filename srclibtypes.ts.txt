// src/lib/types.ts

export type User = {
  id: string
  email: string
  created_at: string
}

export type Post = {
  id: string
  user_id: string
  content: string
  platforms: string[]
  scheduled_at: string | null
  published_at: string | null
  status: 'draft' | 'scheduled' | 'published'
  created_at: string
}

export type Asset = {
  id: string
  user_id: string
  filename: string
  url: string
  type: 'image' | 'video' | 'document'
  size: number
  uploaded_at: string
}

export type Mention = {
  id: string
  user_id: string
  platform: string
  content: string
  author: string
  sentiment: 'positive' | 'neutral' | 'negative'
  processed: boolean
  created_at: string
}

export type AgentLog = {
  id: string
  user_id: string
  agent_type: string
  prompt: string
  response: string
  tokens_used: number
  cost_usd: number
  created_at: string
}

export type AgentResult = {
  content: string
  tokensUsed: number
  cost: number
}
